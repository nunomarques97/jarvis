# jarvis - direct dependencies, pinned versions.
# Install into the repository's own virtual environment, never globally:
#   python -m venv .venv
#   .venv\Scripts\python -m pip install -U pip
#   .venv\Scripts\python -m pip install -r requirements.txt
# The fully resolved lock is in requirements.lock.txt.

# PyTorch CUDA 12.8 wheels: the Windows wheels on PyPI are CPU-only, and older
# CUDA builds have no kernels for the Blackwell architecture (sm_120).
--extra-index-url https://download.pytorch.org/whl/cu128

# Microphone capture and voice activity detection for the resident listener
# (jarvis/ouvido.py).
PyAudio==0.2.14
webrtcvad-wheels==2.0.14

# Local transcription with Whisper, the engine measured against Parakeet and
# the one the environment check tests.
faster-whisper==1.1.1

# NVIDIA Parakeet TDT 0.6B v3 (ONNX), the local STT engine measured against
# faster-whisper by scripts/avaliar_voz.py. The [cpu] extra reuses the CPU
# onnxruntime already in the venv; the model is downloaded separately into
# models/ (see docs/MODELOS.md). Without it the adapter reports itself
# unavailable and is skipped.
onnx-asr[cpu,hub]==0.12.0

# "hey jarvis" wake word
openwakeword==0.6.0

# Local spoken reply. jarvis/voz.py loads the voice once, in process: the
# piper-tts library runs the pt-PT Piper voice. RealtimeTTS (its [piper] extra
# drives the piper.exe shipped in piper-tts) is only used by
# scripts/gerar_wav.py to make test WAVs.
RealtimeTTS[piper]==0.8.5
piper-tts==1.8.0

# Kokoro-82M, the natural English voice. The model files are downloaded
# separately into models/kokoro/ (see docs/MODELOS.md); without them the
# jarvis keeps speaking with the Piper voice.
kokoro-onnx==0.6.1

# Pinned to the CUDA 12.8 build for Blackwell GPUs, so CTranslate2 finds
# cuDNN/cuBLAS inside the venv.
torch==2.9.1+cu128
torchaudio==2.9.1+cu128
