# jarvis - direct dependencies, pinned versions.
# Install into the repository's own virtual environment, never globally:
#   python -m venv .venv
#   .venv\Scripts\python -m pip install -U pip
#   .venv\Scripts\python -m pip install -r requirements.txt
# The fully resolved lock is in requirements.lock.txt.

# PyTorch CUDA 12.8 wheels: the Windows wheels on PyPI are CPU-only, and older
# CUDA builds have no kernels for the Blackwell architecture (sm_120).
--extra-index-url https://download.pytorch.org/whl/cu128

# Base / orchestration: capture, VAD, wake word and STT
RealtimeSTT==0.3.104

# Local transcription on the GPU. Comes as a RealtimeSTT dependency, pinned
# here because it is the engine the environment check tests.
faster-whisper==1.1.1

# "hey jarvis" wake word
openwakeword==0.6.0

# Local spoken reply (0.4.9 had no PiperEngine at all). From 0.7.0 the [piper]
# extra brings the real engine, which drives the piper.exe shipped in the
# piper-tts package.
RealtimeTTS[piper]==0.8.5
piper-tts==1.8.0

# Dependency of RealtimeSTT/openwakeword, pinned to the CUDA 12.8 build for
# Blackwell GPUs and so CTranslate2 finds cuDNN/cuBLAS inside the venv.
torch==2.9.1+cu128
torchaudio==2.9.1+cu128
